    40→
    41→export function LoginView() {
    42→  const goBack = useStore((s) => s.goBack);
    43→
    44→  const { t } = useLang();
    45→  const mounted = useMounted();
    46→  const tr = mounted ? t : (key: any) => (i18nTranslations.id as any)[key] ?? key;
    47→
    48→  const [showPass, setShowPass] = useState(false);
    49→  const [loading, setLoading] = useState(false);
    50→  const [success, setSuccess] = useState(false);
    51→
    52→  const [lEmail, setLEmail] = useState("");
    53→  const [lPass, setLPass] = useState("");
    54→
    55→  const [rName, setRName] = useState("");
    56→  const [rEmail, setREmail] = useState("");
    57→  const [rPhone, setRPhone] = useState("");
    58→  const [rPass, setRPass] = useState("");
    59→  const [rPass2, setRPass2] = useState("");
    60→  const [agree, setAgree] = useState(false);
    61→  const [tab, setTab] = useState<"login" | "register">("login");
    62→  const [forgotOpen, setForgotOpen] = useState(false);
    63→
    64→  // --- Register OTP state ---
    65→  const [rOtp, setROtp] = useState("");
    66→  const [rOtpSending, setROtpSending] = useState(false);
    67→  const [rOtpVerifying, setROtpVerifying] = useState(false);
    68→  const [rOtpVerified, setROtpVerified] = useState(false);
    69→  const [rOtpDevCode, setROtpDevCode] = useState<string | null>(null);
    70→  const [rOtpCooldown, setROtpCooldown] = useState(0);
    71→
    72→  // --- Register email availability check ---
    73→  // Status: "idle" | "checking" | "available" | "taken" | "invalid"
    74→  const [rEmailStatus, setREmailStatus] = useState<"idle" | "checking" | "available" | "taken" | "invalid">("idle");
    75→
    76→  useEffect(() => {
    77→    if (rOtpCooldown <= 0) return;
    78→    const id = setTimeout(() => setROtpCooldown((c) => c - 1), 1000);
    79→    return () => clearTimeout(id);
    80→  }, [rOtpCooldown]);
    81→
    82→  // Debounced email availability check — fires 500ms after the user stops typing.
    83→  // Only checks well-formed emails to avoid pointless API calls.
    84→  useEffect(() => {
    85→    const email = rEmail.trim();
    86→    const emailRegex = /^[^\s@]+@[^\s@]+\.[^\s@]+$/;
    87→    if (!email) {
    88→      setREmailStatus("idle");
    89→      return;
    90→    }
    91→    if (!emailRegex.test(email)) {
    92→      setREmailStatus("invalid");
    93→      return;
    94→    }
    95→    setREmailStatus("checking");
    96→    let cancelled = false;
    97→    const timer = setTimeout(async () => {
    98→      try {
    99→        const res = await fetch(`/api/auth/check-email?email=${encodeURIComponent(email)}`);
   100→        const data = await res.json();
   101→        if (cancelled) return;
   102→        setREmailStatus(data?.exists ? "taken" : "available");
   103→      } catch {
   104→        if (!cancelled) setREmailStatus("idle");
   105→      }
   106→    }, 500);
   107→    return () => {
   108→      cancelled = true;
   109→      clearTimeout(timer);
   110→    };
   111→  }, [rEmail]);
   112→
   113→  // --- Login method toggle (Email vs No. WhatsApp) ---
   114→  const [loginMethod, setLoginMethod] = useState<"email" | "wa">("email");
   115→
   116→  // --- WhatsApp login OTP state (login tab) ---
   117→  const [lWaPhone, setLWaPhone] = useState("");
   118→  const [lWaOtp, setLWaOtp] = useState("");
   119→  const [lWaOtpSending, setLWaOtpSending] = useState(false);
   120→  const [lWaOtpVerifying, setLWaOtpVerifying] = useState(false);
   121→  const [lWaOtpVerified, setLWaOtpVerified] = useState(false);
   122→  const [lWaOtpDevCode, setLWaOtpDevCode] = useState<string | null>(null);
   123→  const [lWaCooldown, setLWaCooldown] = useState(0);
   124→
   125→  useEffect(() => {
   126→    if (lWaCooldown <= 0) return;
   127→    const id = setTimeout(() => setLWaCooldown((c) => c - 1), 1000);
   128→    return () => clearTimeout(id);
   129→  }, [lWaCooldown]);
   130→
   131→  // When phone changes after verification, reset verified state so user must re-verify.
   132→  const prevLWaPhoneRef = useRef<string>("");
   133→  useEffect(() => {
   134→    if (prevLWaPhoneRef.current !== lWaPhone) {
   135→      prevLWaPhoneRef.current = lWaPhone;
   136→      if (lWaOtpVerified) setLWaOtpVerified(false);
   137→    }
   138→  }, [lWaPhone, lWaOtpVerified]);
   139→
   140→  const sendLoginOtp = async () => {
   141→    if (!lWaPhone.trim()) {
   142→      toast.error(tr("errPhoneRequired"));
   143→      return;
   144→    }
   145→    setLWaOtpSending(true);
   146→    setLWaOtpDevCode(null);
   147→    try {
   148→      const res = await fetch("/api/auth/otp", {
   149→        method: "POST",
   150→        headers: { "Content-Type": "application/json" },
   151→        body: JSON.stringify({ action: "send", phone: lWaPhone }),
   152→      });
   153→      const data = await res.json();
   154→      if (!res.ok) {
   155→        // If the number is not registered, show a specific helpful error.
   156→        const isNotFound = /tidak ditemukan|not found|not registered|tidak terdaftar/i.test(
   157→          data.error || ""
   158→        );
   159→        toast.error(isNotFound ? tr("loginWaNotFound") : (data.error || tr("errConnection")));
   160→        return;
   161→      }
   162→      toast.success(data.message || tr("loginWaOtpSentEmail"));
   163→      if (data._devCode) {
   164→        setLWaOtpDevCode(data._devCode);
   165→      }
   166→      setLWaCooldown(60);
   167→    } catch {
   168→      toast.error(tr("errConnection"));
   169→    } finally {
   170→      setLWaOtpSending(false);
   171→    }
   172→  };
   173→
   174→  const verifyLoginOtp = async () => {
   175→    if (lWaOtp.length < 6) {
   176→      toast.error(tr("regOtpRequired"));
   177→      return;
   178→    }
   179→    setLWaOtpVerifying(true);
   180→    try {
   181→      const res = await fetch("/api/auth/otp", {
   182→        method: "POST",
   183→        headers: { "Content-Type": "application/json" },
   184→        body: JSON.stringify({ action: "verify", phone: lWaPhone, code: lWaOtp }),
   185→      });
   186→      const data = await res.json();
   187→      if (!res.ok) {
   188→        toast.error(data.error || tr("regOtpInvalid"));
   189→        return;
   190→      }
   191→      toast.success(tr("loginWaVerified"));
   192→      setLWaOtpVerified(true);
   193→    } catch {
   194→      toast.error(tr("errConnection"));
   195→    } finally {
   196→      setLWaOtpVerifying(false);
   197→    }
   198→  };
   199→
   200→  const doLogin = async (e: React.FormEvent) => {
   201→    e.preventDefault();
   202→
   203→    // ===== WhatsApp OTP login path =====
   204→    if (loginMethod === "wa") {
   205→      if (!lWaPhone.trim()) {
   206→        toast.error(tr("errPhoneRequired"));
   207→        return;
   208→      }
   209→      if (!lWaOtpVerified) {
   210→        toast.error(tr("loginWaOtpFirst"));
   211→        return;
   212→      }
   213→      setLoading(true);
   214→      try {
   215→        const res = await fetch("/api/auth/login", {
   216→          method: "POST",
   217→          headers: { "Content-Type": "application/json" },
   218→          body: JSON.stringify({ phone: lWaPhone }),
   219→        });
   220→        const data = await res.json();
   221→        if (!res.ok) {
   222→          const isNotFound = /tidak terdaftar|not registered|not found/i.test(data.error || "");
   223→          toast.error(isNotFound ? tr("loginWaNotFound") : (data.error || tr("errLogin")));

   225→        }
   226→        const setUser = useStore.getState().setUser;
   227→        const goHome = useStore.getState().goHome;
   228→        const goToAdmin = useStore.getState().goToAdmin;

   231→        toast.success(formatT(tr("welcomeBack"), { name: data.user.name }));
   232→        const isAdmin = data.user.role === "admin" || data.user.role === "superadmin";
   233→        setTimeout(() => isAdmin ? goToAdmin() : goHome(), 900);
   234→      } catch {
   235→        toast.error(tr("errConnection"));
   236→      } finally {
   237→        setLoading(false);
   238→      }

   241→
   242→    // ===== Email + password login path (default) =====
   243→    if (!lEmail.trim() || !lPass) {
   244→      toast.error(tr("errEmailPass"));
   245→      return;
   246→    }
   247→    setLoading(true);
   248→    try {
   249→      const res = await fetch("/api/auth/login", {
   250→        method: "POST",
   251→        headers: { "Content-Type": "application/json" },
   252→        body: JSON.stringify({ email: lEmail, password: lPass }),
   253→      });
   254→      const data = await res.json();
   255→      if (!res.ok) {
   256→        toast.error(data.error || tr("errLogin"));
   257→        return;
   258→      }
   259→      const setUser = useStore.getState().setUser;
   260→      const goHome = useStore.getState().goHome;
   261→      const goToAdmin = useStore.getState().goToAdmin;
   262→      setUser(data.user);
   263→      setSuccess(true);
   264→      toast.success(formatT(tr("welcomeBack"), { name: data.user.name }));
   265→      const isAdmin = data.user.role === "admin" || data.user.role === "superadmin";
   266→      setTimeout(() => isAdmin ? goToAdmin() : goHome(), 900);
   267→    } catch {
   268→      toast.error(tr("errConnection"));
   269→    } finally {
   270→      setLoading(false);
   271→    }
   272→  };
   273→
   274→  const sendRegOtp = async () => {
   275→    if (!rPhone.trim()) {
   276→      toast.error(tr("regOtpPhoneFirst"));
   277→      return;
   278→    }
   279→    setROtpSending(true);
   280→    setROtpDevCode(null);
   281→    try {
   282→      const res = await fetch("/api/auth/register-otp", {
   283→        method: "POST",
   284→        headers: { "Content-Type": "application/json" },
   285→        body: JSON.stringify({ action: "send", phone: rPhone }),
   286→      });
   287→      const data = await res.json();
   288→      if (!res.ok) {
   289→        toast.error(data.error || tr("errConnection"));
   290→        return;
   291→      }
   292→      toast.success(data.message || tr("regOtpSent"));
   293→      if (data._devCode) {
   294→        setROtpDevCode(data._devCode);
   295→      }
   296→      setROtpCooldown(60);
   297→    } catch {
   298→      toast.error(tr("errConnection"));
   299→    } finally {
   300→      setROtpSending(false);
   301→    }
   302→  };
   303→
   304→  const verifyRegOtp = async () => {
   305→    if (rOtp.length < 6) {
   306→      toast.error(tr("regOtpRequired"));
   307→      return;
   308→    }
   309→    setROtpVerifying(true);
   310→    try {
   311→      const res = await fetch("/api/auth/register-otp", {
   312→        method: "POST",
   313→        headers: { "Content-Type": "application/json" },
   314→        body: JSON.stringify({ action: "verify", phone: rPhone, code: rOtp }),
   315→      });
   316→      const data = await res.json();
   317→      if (!res.ok) {
   318→        toast.error(data.error || tr("regOtpInvalid"));
   319→        return;
   320→      }
   321→      toast.success(tr("otpVerified"));
   322→      setROtpVerified(true);
   323→    } catch {
   324→      toast.error(tr("errConnection"));
   325→    } finally {
   326→      setROtpVerifying(false);
   327→    }
   328→  };
   329→
   330→  // When phone changes after verification, reset verified state so user must re-verify.
   331→  const prevPhoneRef = useRef<string>("");
   332→  useEffect(() => {
   333→    if (prevPhoneRef.current !== rPhone) {
   334→      prevPhoneRef.current = rPhone;
   335→      if (rOtpVerified) setROtpVerified(false);
   336→    }
   337→  }, [rPhone, rOtpVerified]);
   338→
   339→  const doRegister = async (e: React.FormEvent) => {
   340→    e.preventDefault();
   341→    if (!rName.trim() || !rEmail.trim() || !rPass) {
   342→      toast.error(tr("errRequired"));
   343→      return;
   344→    }
   345→    if (rEmailStatus === "invalid") {
   346→      toast.error(tr("emailInvalid"));
   347→      return;
   348→    }
   349→    if (rEmailStatus === "taken") {
   350→      toast.error(tr("errEmailTaken"));
   351→      return;
   352→    }
   353→    // If still checking, wait for it to resolve before allowing registration
   354→    // so we never submit a duplicate even in a race.
   355→    if (rEmailStatus === "checking") {
   356→      toast.error(tr("emailChecking"));
   357→      return;
   358→    }
   359→    if (!rOtpVerified) {
   360→      toast.error(tr("errPhoneNotVerified"));
   361→      return;
   362→    }
   363→    if (rPass.length < 6) {
   364→      toast.error(tr("errPassLength"));
   365→      return;
   366→    }
   367→    if (rPass !== rPass2) {
   368→      toast.error(tr("errPassMatch"));
   369→      return;
   370→    }
   371→    if (!agree) {
   372→      toast.error(tr("errAgree"));
   373→      return;
   374→    }
   375→    setLoading(true);
   376→    try {
   377→      const res = await fetch("/api/auth/register", {
   378→        method: "POST",
   379→        headers: { "Content-Type": "application/json" },
   380→        body: JSON.stringify({
   381→          name: rName,
   382→          email: rEmail,
   383→          password: rPass,
   384→          phone: rPhone,
   385→        }),
   386→      });
   387→      const data = await res.json();
   388→      if (!res.ok) {
   389→        toast.error(data.error || tr("errRegister"));
   390→        return;
   391→      }
   392→      const setUser = useStore.getState().setUser;
   393→      const goToPost = useStore.getState().goToPost;
   394→      const goToAdmin = useStore.getState().goToAdmin;
   395→      setUser(data.user);
   396→      setSuccess(true);
   397→      toast.success(tr("registerSuccess"));
   398→      const isAdmin = data.user.role === "admin" || data.user.role === "superadmin";
   399→      setTimeout(() => isAdmin ? goToAdmin() : goToPost(), 1100);
   400→    } catch {
   401→      toast.error(tr("errConnection"));
   402→    } finally {
   403→      setLoading(false);
   404→    }
   405→  };
   406→
   407→  if (success) {
   408→    return (
   409→      <div className="mx-auto flex max-w-md flex-col items-center px-4 py-20 text-center animate-fade-up">
   410→        <div className="grid size-20 place-items-center rounded-full bg-primary/10">
   411→          <CheckCircle2 className="size-12 text-primary" />
   412→        </div>
   413→        <h2 className="mt-4 text-2xl font-bold">{tr("loginSuccess")}</h2>
   414→        <p className="mt-2 text-sm text-muted-foreground">
   415→          {tr("loginRedirect")}
   416→        </p>
   417→        <Loader2 className="mt-4 size-5 animate-spin text-primary" />
   418→      </div>
   419→    );
   420→  }
   421→
   422→  return (
   423→    <div className="animate-fade-up">
   424→      {/* ===== MOBILE: compact single column ===== */}
   425→      <div className="flex min-h-[calc(100vh-4rem)] flex-col px-4 py-6 md:hidden">
   426→        <button
   427→          onClick={goBack}
   428→          className="mb-4 inline-flex w-fit items-center gap-1 text-sm text-muted-foreground hover:text-primary"
   429→        >
   430→          <ChevronLeft className="size-4" /> {tr("back")}
   431→        </button>
   432→        <div className="flex flex-col items-center text-center">
   433→          {/* eslint-disable-next-line @next/next/no-img-element */}
   434→          <img src="/logo.jpeg" alt="mesinKU" className="size-16 rounded-2xl shadow-lg object-cover" />
   435→          <h1 className="mt-3 text-2xl font-extrabold tracking-tight">
   436→            <span className="text-primary">mesin</span>KU
   437→          </h1>
   438→          <p className="mt-1 text-sm text-muted-foreground">{tr("loginTitle")}</p>
   439→        </div>
   440→        <div className="mt-6 w-full max-w-md mx-auto">
   441→          <FormSection
   442→            tab={tab} setTab={setTab}
   443→            showPass={showPass} setShowPass={setShowPass}
   444→            loading={loading}
   445→            lEmail={lEmail} setLEmail={setLEmail}
   446→            lPass={lPass} setLPass={setLPass}
   447→            loginMethod={loginMethod} setLoginMethod={setLoginMethod}
   448→            lWaPhone={lWaPhone} setLWaPhone={setLWaPhone}
   449→            lWaOtp={lWaOtp} setLWaOtp={setLWaOtp}
   450→            lWaOtpSending={lWaOtpSending}
   451→            lWaOtpVerifying={lWaOtpVerifying}
   452→            lWaOtpVerified={lWaOtpVerified}
   453→            lWaOtpDevCode={lWaOtpDevCode}
   454→            lWaCooldown={lWaCooldown}
   455→            sendLoginOtp={sendLoginOtp}
   456→            verifyLoginOtp={verifyLoginOtp}
   457→            rName={rName} setRName={setRName}
   458→            rEmail={rEmail} setREmail={setREmail}
   459→            rEmailStatus={rEmailStatus}
   460→            rPhone={rPhone} setRPhone={setRPhone}
   461→            rPass={rPass} setRPass={setRPass}
   462→            rPass2={rPass2} setRPass2={setRPass2}
   463→            agree={agree} setAgree={setAgree}
   464→            rOtp={rOtp} setROtp={setROtp}
   465→            rOtpSending={rOtpSending}
   466→            rOtpVerifying={rOtpVerifying}
   467→            rOtpVerified={rOtpVerified}
   468→            rOtpDevCode={rOtpDevCode}
   469→            rOtpCooldown={rOtpCooldown}
   470→            sendRegOtp={sendRegOtp}
   471→            verifyRegOtp={verifyRegOtp}
   472→            doLogin={doLogin} doRegister={doRegister}
   473→            onForgotPassword={() => setForgotOpen(true)}
   474→            tr={tr}
   475→          />
   476→        </div>
   477→      </div>
   478→
   479→      {/* ===== DESKTOP: 1/2 orange block + 1/2 centered form ===== */}
   480→      <div className="hidden md:grid md:grid-cols-2 md:min-h-[calc(100vh-4rem)]">
   481→        {/* LEFT: Orange block with logo + marketing text */}
   482→        <div className="relative flex flex-col items-center justify-center bg-primary px-12 py-16 overflow-hidden">
   483→          {/* Decorative circles */}
   484→          <div className="absolute inset-0 opacity-10">
   485→            <div className="absolute -top-24 -left-24 size-72 rounded-full bg-white/20" />
   486→            <div className="absolute -bottom-32 -right-32 size-96 rounded-full bg-white/10" />
   487→            <div className="absolute top-1/2 left-1/2 -translate-x-1/2 -translate-y-1/2 size-[600px] rounded-full bg-white/5" />
   488→          </div>
   489→          <div className="relative z-10 flex flex-col items-center text-center max-w-sm">
   490→            {/* eslint-disable-next-line @next/next/no-img-element */}
   491→            <img
   492→              src="/logo.jpeg"
   493→              alt="mesinKU"
   494→              className="size-24 rounded-3xl shadow-2xl object-cover ring-4 ring-white/20"
   495→            />
   496→            <h1 className="mt-6 text-3xl font-extrabold tracking-tight text-primary-foreground">

   507→              {[
   508→                { icon: ShieldCheck, label: "Seller Terverifikasi" },
   509→                { icon: MapPin, label: "Se-Indonesia" },
   510→                { icon: Cog, label: "Mesin Berkualitas" },
   511→              ].map(({ icon: Icon, label }) => (
   512→                <span
   513→                  key={label}

   515→                >
   516→                  <Icon className="size-3.5" />
   517→                  {label}
   518→                </span>
   519→              ))}
   520→            </div>
   521→          </div>
   522→        </div>
   523→
   524→        {/* RIGHT: Centered login/register form */}
   525→        <div className="flex flex-col items-center justify-center px-8 py-12">
   526→          <div className="w-full max-w-md">
   527→            <button
   528→              onClick={goBack}
