   196→      ? "Iklan ditolak. Boost hanya tersedia untuk iklan yang sudah aktif."

   272→        method: "PATCH",
   273→        headers: { "Content-Type": "application/json" },
   274→        body: JSON.stringify({
   275→          package: selectedPackage,
   276→          paymentMethod: needsPayment ? paymentMethod : undefined,
   277→          uniqueCode: uniqueCode !== null ? uniqueCode : undefined,

   293→      toast.error(e.message || "Gagal mengaktifkan paket");
   294→    } finally {
   295→      setSubmitting(false);
   296→    }
   297→  };

   300→    <>
   301→    {/* ===== UPGRADE PAKET PAGE (fullscreen, 2 kolom di desktop) ===== */}

  1099→/**
  1100→ * Sanitize user name/email to be filesystem-safe.
  1101→ * "John Doe" → "John_Doe"
  1102→ * "user@email.com" → "user_email.com"
  1103→ * Removes special chars that break URLs/filenames.
  1104→ */
  1105→function sanitizeFileName(name: string): string {
  1106→  return name
  1107→    .trim()
  1108→    .replace(/\s+/g, "_")           // spaces → underscore
  1109→    .replace(/[@/\\?%*:|"<>]/g, "_") // special chars → underscore
  1110→    .replace(/_+/g, "_")              // collapse multiple underscores
  1111→    .substring(0, 50);               // max 50 chars
  1112→}
