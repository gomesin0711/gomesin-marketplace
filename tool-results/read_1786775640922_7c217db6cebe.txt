    45→    45→export async function GET(req: NextRequest) {

    51→    51→  const maxPrice = searchParams.get("maxPrice");
    52→    52→  const province = searchParams.get("province") || "";
    53→    53→  const packageType = searchParams.get("packageType") || "";
    54→    54→  const city = searchParams.get("city")?.trim() || "";
    55→    55→  const sellerName = searchParams.get("seller")?.trim() || "";
    56→    56→  const sort = searchParams.get("sort") || "newest";
    57→    57→  const featuredOnly = searchParams.get("featured") === "1";
    58→    58→  const idsParam = searchParams.get("ids");
    59→    59→  const ids = idsParam ? idsParam.split(",").filter(Boolean) : null;
    60→    60→  const page = Math.max(1, parseInt(searchParams.get("page") || "1", 10));
    61→    61→  const limit = Math.min(48, Math.max(1, parseInt(searchParams.get("limit") || "24", 10)));
    62→    62→  const weekOnly = searchParams.get("week") === "1";
    63→    63→
    64→    64→  try {
    65→    65→    const where: Record<string, any> = { status: "active", paymentStatus: "paid", violationFlag: false };
    66→    66→    if (ids && ids.length) {
    67→    67→      where.id = { in: ids };
    68→    68→    }
    69→    69→
    70→    70→    if (q) {
    71→    71→      where.OR = [
    72→    72→        { title: { contains: q } },
    73→    73→        { description: { contains: q } },
    74→    74→        { brand: { contains: q } },
    75→    75→        { seller: { name: { contains: q } } },

    81→    81→      // (iklan jasa pakai condition="jasa", bukan categoryId jasa-teknisi).
    82→    82→      if (category === "jasa-teknisi") {
    83→    83→        where.condition = "jasa";
    84→    84→      } else {
    85→    85→        where.category = { slug: category };
    86→    86→      }
    87→    87→    }
    88→    88→    if (condition) where.condition = condition;
    89→    89→    if (province) where.province = province;
    90→    90→    if (city) where.city = { contains: city };
    91→    91→    if (sellerName) where.seller = { name: { contains: sellerName } };
    92→    92→    if (minPrice || maxPrice) {
    93→    93→      where.price = {};
    94→    94→      if (minPrice) where.price.gte = Math.floor(Number(minPrice));
    95→    95→      if (maxPrice) where.price.lte = Math.floor(Number(maxPrice));
    96→    96→    }
    97→    97→    if (featuredOnly) where.featured = true;
    98→    98→    // Week filter: only listings from the last 7 days
    99→    99→    if (weekOnly) {
   100→   100→      const weekAgo = new Date();
   101→   101→      weekAgo.setDate(weekAgo.getDate() - 7);
   102→   102→      where.createdAt = { gte: weekAgo };
   103→   103→    }
   104→   104→    if (packageType) {
   105→   105→      const pkgList = packageType.split(",").map((p) => p.trim()).filter(Boolean);
   106→   106→      if (pkgList.length === 1) where.packageType = pkgList[0];
   107→   107→      else if (pkgList.length > 1) where.packageType = { in: pkgList };
   108→   108→    }
   109→   109→
   110→   110→    const orderBy: Record<string, any> =
   111→   111→      sort === "price-asc"
   112→   112→        ? { price: "asc" }
   113→   113→        : sort === "price-desc"
   114→   114→        ? { price: "desc" }
   115→   115→        : sort === "popular"
   116→   116→        ? { views: "desc" }
   117→   117→        : { createdAt: "desc" };
   118→   118→
   119→   119→    const [total, rows] = await Promise.all([
   120→   120→      db.listing.count({ where }),
   121→   121→      db.listing.findMany({
   122→   122→        where,
   123→   123→        orderBy,
   124→   124→        skip: (page - 1) * limit,
   125→   125→        take: limit,
   126→   126→        include: { category: true, seller: true, user: { select: { id: true, name: true, phone: true, email: true, city: true, logoImage: true, bannerImage: true } } },
   127→   127→      }),
   128→   128→    ]);
   129→   129→
   130→   130→    // Newest ads first — no promo-rank grouping, pure createdAt desc order.
   131→   131→    const listings = rows.map(parseListing);
   132→   132→
   133→   133→    return NextResponse.json({
   134→   134→      listings,
   135→   135→      total,
   136→   136→      page,
   137→   137→      limit,
   138→   138→      totalPages: Math.ceil(total / limit),
   139→   139→    });
   140→   140→  } catch (error) {

   142→   142→    // DO NOT return getFallbackListings here anymore — fall through to Supabase

   144→   144→
   145→   145→  // --- Path B: Vercel (raw Supabase) ---
   146→   146→  // NOTE: Supabase tables in this project have NO foreign-key relationships
   147→   147→  // declared, so the nested .select("*, category(*)") pattern that PostgREST
   148→   148→  // requires FKs for would FAIL and silently return []. We select only "*" and
   149→   149→  // batch-fetch the related Category/Seller/User rows manually (same pattern
   150→   150→  // as /api/admin/listings/route.ts).
   151→   151→  try {
   152→   152→    const supabase = await getSupabase();
   153→   153→
   154→   154→    // Resolve category filter once: either condition="jasa" (for "jasa-teknisi"
   155→   155→    // slug) or categoryId=<resolved id> (looked up by slug). Hoisted so we can
   156→   156→    // re-use the resolved values for the count query below.
   157→   157→    let categoryCondition: string | null = null; // e.g. "jasa"
   158→   158→    let categoryIdFilter: string | null = null;
   159→   159→    if (category) {
   160→   160→      if (category === "jasa-teknisi") {
   161→   161→        categoryCondition = "jasa";
   162→   162→      } else {

   166→   166→          .eq("slug", category)
   167→   167→          .limit(1)
   168→   168→          .single();
   169→   169→        if (cat) categoryIdFilter = cat.id;
   170→   170→      }
   171→   171→    }
   172→   172→
   173→   173→    // Build Supabase query with the SAME filters as the Prisma where clause:
   174→   174→    // - status = "active"
   175→   175→    // - paymentStatus = "paid"
   176→   176→    // - violationFlag = false
   177→   177→    let query = supabase
   178→   178→      .from("Listing")
   179→   179→      .select("*")
   180→   180→      .eq("status", "active")
   181→   181→      .eq("paymentStatus", "paid")
   182→   182→      .eq("violationFlag", false);
   183→   183→
   184→   184→    // Apply filters (mirror the Prisma where clause):
   185→   185→    if (ids && ids.length) query = query.in("id", ids);
   186→   186→    if (q) {
   187→   187→      // Supabase text search: use .or() with ilike for each searchable field.
   188→   188→      // Cannot do cross-table search on seller.name without a join — just
   189→   189→      // search title/description/brand/city (minor edge case skipped).
   190→   190→      query = query.or(
   191→   191→        `title.ilike.%${q}%,description.ilike.%${q}%,brand.ilike.%${q}%,city.ilike.%${q}%`
   192→   192→      );
   193→   193→    }
   194→   194→    if (categoryCondition) query = query.eq("condition", categoryCondition);
   195→   195→    if (categoryIdFilter) query = query.eq("categoryId", categoryIdFilter);
   196→   196→    if (condition && !categoryCondition) query = query.eq("condition", condition);
   197→   197→    if (province) query = query.eq("province", province);
   198→   198→    if (city) query = query.ilike("city", `%${city}%`);
   199→   199→    if (minPrice) query = query.gte("price", Math.floor(Number(minPrice)));
   200→   200→    if (maxPrice) query = query.lte("price", Math.floor(Number(maxPrice)));
   201→   201→    if (featuredOnly) query = query.eq("featured", true);
   202→   202→    if (weekOnly) {
   203→   203→      const weekAgo = new Date();
   204→   204→      weekAgo.setDate(weekAgo.getDate() - 7);
   205→   205→      query = query.gte("createdAt", weekAgo.toISOString());
   206→   206→    }
   207→   207→    if (packageType) {
   208→   208→      const pkgList = packageType.split(",").map((p) => p.trim()).filter(Boolean);
   209→   209→      if (pkgList.length === 1) query = query.eq("packageType", pkgList[0]);
   210→   210→      else if (pkgList.length > 1) query = query.in("packageType", pkgList);
   211→   211→    }
   212→   212→
   213→   213→    // Apply sort (mirror Prisma orderBy)
   214→   214→    if (sort === "price-asc") query = query.order("price", { ascending: true });
   215→   215→    else if (sort === "price-desc") query = query.order("price", { ascending: false });
   216→   216→    else if (sort === "popular") query = query.order("views", { ascending: false });
   217→   217→    else query = query.order("createdAt", { ascending: false });
   218→   218→
   219→   219→    // Apply pagination
   220→   220→    query = query.range((page - 1) * limit, (page - 1) * limit + limit - 1);
   221→   221→
   222→   222→    const { data: rows, error } = await query;
   223→   223→    if (error) {
   224→   224→      console.error("[listings] Supabase GET error:", error);
   225→   225→      const filters: ListingFilters = {
   226→   226→        q: q || undefined,
   227→   227→        category: category || undefined,
   228→   228→        condition: condition || undefined,
   229→   229→        province: province || undefined,
   230→   230→        packageType: packageType || undefined,
   231→   231→        sort: sort || undefined,
   232→   232→        page,
   233→   233→        limit,
   234→   234→        ids,
   235→   235→        featured: featuredOnly || undefined,
   236→   236→      };
   237→   237→      return NextResponse.json(getFallbackListings(filters));
   238→   238→    }
   239→   239→
   240→   240→    const finalRows: any[] = rows || [];
   241→   241→
   242→   242→    // Batch-fetch related rows by their IDs (single round-trip per table).
   243→   243→    const categoryIds = [...new Set(finalRows.map((r: any) => r.categoryId).filter(Boolean))];
   244→   244→    const sellerIds = [...new Set(finalRows.map((r: any) => r.sellerId).filter(Boolean))];
   245→   245→    const userIds = [...new Set(finalRows.map((r: any) => r.userId).filter(Boolean))];
   246→   246→
   247→   247→    const [categoriesRes, sellersRes, usersRes] = await Promise.all([
   248→   248→      categoryIds.length
   249→   249→        ? supabase.from("Category").select("*").in("id", categoryIds)
   250→   250→        : Promise.resolve({ data: [], error: null }),
   251→   251→      sellerIds.length
   252→   252→        ? supabase.from("Seller").select("*").in("id", sellerIds)
   253→   253→        : Promise.resolve({ data: [], error: null }),
   254→   254→      userIds.length
   255→   255→        ? supabase
   256→   256→            .from("User")
   257→   257→            .select("id, name, phone, email, city, logoImage, bannerImage")
   258→   258→            .in("id", userIds)
   259→   259→        : Promise.resolve({ data: [], error: null }),
   260→   260→    ]);
   261→   261→
   262→   262→    const categoryMap = new Map((categoriesRes.data || []).map((c: any) => [c.id, c]));
   263→   263→    const sellerMap = new Map((sellersRes.data || []).map((s: any) => [s.id, s]));
   264→   264→    const userMap = new Map((usersRes.data || []).map((u: any) => [u.id, u]));
   265→   265→
   266→   266→    const listings = finalRows.map((row: any) => {
   267→   267→      const withRelations = {
   268→   268→        ...row,
   269→   269→        category: categoryMap.get(row.categoryId) ?? null,
   270→   270→        seller: sellerMap.get(row.sellerId) ?? null,
   271→   271→        user: userMap.get(row.userId) ?? null,
   272→   272→      };
   273→   273→      return parseSupabaseListing(withRelations);
   274→   274→    });
   275→   275→
   276→   276→    // Get total count via a separate count query (re-apply same filters).
   277→   277→    // Supabase's .range() with .select("*") does not return total count;
   278→   278→    // we use { count: "exact", head: true } to get the count without data.
   279→   279→    let countQuery = supabase
   280→   280→      .from("Listing")
   281→   281→      .select("id", { count: "exact", head: true })
   282→   282→      .eq("status", "active")
   283→   283→      .eq("paymentStatus", "paid")
   284→   284→      .eq("violationFlag", false);
   285→   285→    if (ids && ids.length) countQuery = countQuery.in("id", ids);
   286→   286→    if (categoryCondition) countQuery = countQuery.eq("condition", categoryCondition);
   287→   287→    if (categoryIdFilter) countQuery = countQuery.eq("categoryId", categoryIdFilter);
   288→   288→    if (condition && !categoryCondition) countQuery = countQuery.eq("condition", condition);
   289→   289→    if (province) countQuery = countQuery.eq("province", province);
   290→   290→    if (city) countQuery = countQuery.ilike("city", `%${city}%`);
   291→   291→    if (minPrice) countQuery = countQuery.gte("price", Math.floor(Number(minPrice)));
   292→   292→    if (maxPrice) countQuery = countQuery.lte("price", Math.floor(Number(maxPrice)));
   293→   293→    if (featuredOnly) countQuery = countQuery.eq("featured", true);
   294→   294→    if (weekOnly) {
   295→   295→      const weekAgo = new Date();
   296→   296→      weekAgo.setDate(weekAgo.getDate() - 7);
   297→   297→      countQuery = countQuery.gte("createdAt", weekAgo.toISOString());
   298→   298→    }
   299→   299→    if (packageType) {
   300→   300→      const pkgList = packageType.split(",").map((p) => p.trim()).filter(Boolean);
   301→   301→      if (pkgList.length === 1) countQuery = countQuery.eq("packageType", pkgList[0]);
   302→   302→      else if (pkgList.length > 1) countQuery = countQuery.in("packageType", pkgList);
   303→   303→    }
   304→   304→    if (q)
   305→   305→      countQuery = countQuery.or(
   306→   306→        `title.ilike.%${q}%,description.ilike.%${q}%,brand.ilike.%${q}%,city.ilike.%${q}%`
   307→   307→      );
   308→   308→
   309→   309→    const { count: totalCount } = await countQuery;
   310→   310→    const total = totalCount || listings.length;
   311→   311→
   312→   312→    return NextResponse.json({
   313→   313→      listings,
   314→   314→      total,
   315→   315→      page,
   316→   316→      limit,
   317→   317→      totalPages: Math.ceil(total / limit),
   318→   318→    });
   319→   319→  } catch (error) {
   320→   320→    console.error("GET /api/listings Supabase error, falling back to seed data:", error);
   321→   321→    const filters: ListingFilters = {
   322→   322→      q: q || undefined,
   323→   323→      category: category || undefined,
   324→   324→      condition: condition || undefined,
   325→   325→      province: province || undefined,
   326→   326→      packageType: packageType || undefined,
   327→   327→      sort: sort || undefined,
   328→   328→      page,
   329→   329→      limit,
   330→   330→      ids,
   331→   331→      featured: featuredOnly || undefined,
   332→   332→    };
   333→   333→    return NextResponse.json(getFallbackListings(filters));
   334→   334→  }
   335→   335→}

   404→   404→

   406→   406→        // Also validate that the provided categoryId actually EXISTS in the DB.
