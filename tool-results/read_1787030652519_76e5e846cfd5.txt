    42→export function LoginView() {
    43→  const goBack = useStore((s) => s.goBack);
    44→
    45→  const { t } = useLang();
    46→  const mounted = useMounted();
    47→  const tr = mounted ? t : (key: any) => (i18nTranslations.id as any)[key] ?? key;
    48→
    49→  const [showPass, setShowPass] = useState(false);
    50→  const [loading, setLoading] = useState(false);
    51→  const [success, setSuccess] = useState(false);
    52→
    53→  const [lEmail, setLEmail] = useState("");
    54→  const [lPass, setLPass] = useState("");
    55→
    56→  const [rName, setRName] = useState("");
    57→  const [rEmail, setREmail] = useState("");
    58→  const [rPhone, setRPhone] = useState("");
    59→  const [rPass, setRPass] = useState("");
    60→  const [rPass2, setRPass2] = useState("");
    61→  const [agree, setAgree] = useState(false);
    62→  const [tab, setTab] = useState<"login" | "register">("login");
    63→  const [forgotOpen, setForgotOpen] = useState(false);
    64→
    65→  // --- Register OTP state ---
    66→  const [rOtp, setROtp] = useState("");
    67→  const [rOtpSending, setROtpSending] = useState(false);
    68→  const [rOtpVerifying, setROtpVerifying] = useState(false);
    69→  const [rOtpVerified, setROtpVerified] = useState(false);
    70→  const [rOtpDevCode, setROtpDevCode] = useState<string | null>(null);
    71→  const [rOtpCooldown, setROtpCooldown] = useState(0);
    72→
    73→  // --- Register email availability check ---
    74→  // Status: "idle" | "checking" | "available" | "taken" | "invalid"
    75→  const [rEmailStatus, setREmailStatus] = useState<"idle" | "checking" | "available" | "taken" | "invalid">("idle");
    76→
    77→  // --- Register name & phone availability checks ---
    78→  const [rNameStatus, setRNameStatus] = useState<"idle" | "checking" | "available" | "taken">("idle");
    79→  const [rPhoneStatus, setRPhoneStatus] = useState<"idle" | "checking" | "available" | "taken">("idle");
    80→
    81→  useEffect(() => {
    82→    if (rOtpCooldown <= 0) return;
    83→    const id = setTimeout(() => setROtpCooldown((c) => c - 1), 1000);
    84→    return () => clearTimeout(id);
    85→  }, [rOtpCooldown]);
    86→
    87→  // Debounced email availability check — fires 500ms after the user stops typing.
    88→  // Only checks well-formed emails to avoid pointless API calls.
    89→  useEffect(() => {
    90→    const email = rEmail.trim();
    91→    const emailRegex = /^[^\s@]+@[^\s@]+\.[^\s@]+$/;
    92→    if (!email) {
    93→      setREmailStatus("idle");
    94→      return;
    95→    }
    96→    if (!emailRegex.test(email)) {
    97→      setREmailStatus("invalid");
    98→      return;
    99→    }
   100→    setREmailStatus("checking");
   101→    let cancelled = false;
   102→    const timer = setTimeout(async () => {
   103→      try {
   104→        const res = await fetch(`/api/auth/check-email?email=${encodeURIComponent(email)}`);
   105→        const data = await res.json();
   106→        if (cancelled) return;
   107→        setREmailStatus(data?.exists ? "taken" : "available");
   108→      } catch {
   109→        if (!cancelled) setREmailStatus("idle");
   110→      }
   111→    }, 500);
   112→    return () => {
   113→      cancelled = true;
   114→      clearTimeout(timer);
   115→    };
   116→  }, [rEmail]);
   117→
   118→  // Debounced NAME availability check — fires 500ms after the user stops typing.
   119→  // Blocks registration if the name is already taken (case-insensitive).
   120→  useEffect(() => {
   121→    const name = rName.trim();
   122→    if (name.length < 2) {
   123→      setRNameStatus("idle");
   124→      return;
   125→    }
   126→    setRNameStatus("checking");
   127→    let cancelled = false;
   128→    const timer = setTimeout(async () => {
   129→      try {
   130→        const res = await fetch(`/api/auth/check-availability?name=${encodeURIComponent(name)}`);
   131→        const data = await res.json();
   132→        if (cancelled) return;
   133→        setRNameStatus(data?.nameTaken ? "taken" : "available");
   134→      } catch {
   135→        if (!cancelled) setRNameStatus("idle");
   136→      }
   137→    }, 500);
   138→    return () => {
   139→      cancelled = true;
   140→      clearTimeout(timer);
   141→    };
   142→  }, [rName]);
   143→
   144→  // Debounced PHONE (WhatsApp) availability check — fires 500ms after the user
   145→  // stops typing. Only checks once the number has at least 9 digits.
   146→  useEffect(() => {
   147→    const phone = rPhone.trim();
   148→    const digits = phone.replace(/[^0-9]/g, "");
   149→    if (digits.length < 9) {
   150→      setRPhoneStatus("idle");
   151→      return;
   152→    }
   153→    setRPhoneStatus("checking");
   154→    let cancelled = false;
   155→    const timer = setTimeout(async () => {
   156→      try {
   157→        const res = await fetch(`/api/auth/check-availability?phone=${encodeURIComponent(phone)}`);
   158→        const data = await res.json();
   159→        if (cancelled) return;
   160→        setRPhoneStatus(data?.phoneTaken ? "taken" : "available");
   161→      } catch {
   162→        if (!cancelled) setRPhoneStatus("idle");
   163→      }
   164→    }, 500);
   165→    return () => {
   166→      cancelled = true;
   167→      clearTimeout(timer);
   168→    };
   169→  }, [rPhone]);
   170→
   171→  // --- Login method toggle (Email vs No. WhatsApp) ---
   172→  const [loginMethod, setLoginMethod] = useState<"email" | "wa">("email");
   173→
   174→  // --- WhatsApp login OTP state (login tab) ---
   175→  const [lWaPhone, setLWaPhone] = useState("");
   176→  const [lWaOtp, setLWaOtp] = useState("");
   177→  const [lWaOtpSending, setLWaOtpSending] = useState(false);
   178→  const [lWaOtpVerifying, setLWaOtpVerifying] = useState(false);
   179→  const [lWaOtpVerified, setLWaOtpVerified] = useState(false);
   180→  const [lWaOtpDevCode, setLWaOtpDevCode] = useState<string | null>(null);
   181→  const [lWaCooldown, setLWaCooldown] = useState(0);
   182→
   183→  useEffect(() => {
   184→    if (lWaCooldown <= 0) return;
   185→    const id = setTimeout(() => setLWaCooldown((c) => c - 1), 1000);
   186→    return () => clearTimeout(id);
   187→  }, [lWaCooldown]);
   188→
   189→  // When phone changes after verification, reset verified state so user must re-verify.
   190→  const prevLWaPhoneRef = useRef<string>("");
   191→  useEffect(() => {
   192→    if (prevLWaPhoneRef.current !== lWaPhone) {
   193→      prevLWaPhoneRef.current = lWaPhone;
   194→      if (lWaOtpVerified) setLWaOtpVerified(false);
   195→    }
   196→  }, [lWaPhone, lWaOtpVerified]);
   197→
   198→  const sendLoginOtp = async () => {
   199→    if (!lWaPhone.trim()) {
   200→      toast.error(tr("errPhoneRequired"));
   201→      return;
   202→    }
   203→    setLWaOtpSending(true);
   204→    setLWaOtpDevCode(null);
   205→    try {
   206→      const res = await fetch("/api/auth/otp", {
   207→        method: "POST",
   208→        headers: { "Content-Type": "application/json" },
   209→        body: JSON.stringify({ action: "send", phone: lWaPhone }),
   210→      });
   211→      const data = await res.json();
   212→      if (!res.ok) {
   213→        // If the number is not registered, show a specific helpful error.
   214→        const isNotFound = /tidak ditemukan|not found|not registered|tidak terdaftar/i.test(
   215→          data.error || ""
   216→        );
   217→        toast.error(isNotFound ? tr("loginWaNotFound") : (data.error || tr("errConnection")));
   218→        return;
   219→      }
   220→      toast.success(data.message || tr("loginWaOtpSentEmail"));
   221→      if (data._devCode) {
   222→        setLWaOtpDevCode(data._devCode);
   223→      }
   224→      setLWaCooldown(60);
   225→    } catch {
   226→      toast.error(tr("errConnection"));
   227→    } finally {
   228→      setLWaOtpSending(false);
   229→    }
   230→  };
   231→
   232→  const verifyLoginOtp = async () => {
   233→    if (lWaOtp.length < 6) {
   234→      toast.error(tr("regOtpRequired"));
   235→      return;
   236→    }
   237→    setLWaOtpVerifying(true);
   238→    try {
   239→      const res = await fetch("/api/auth/otp", {
   240→        method: "POST",
   241→        headers: { "Content-Type": "application/json" },
   242→        body: JSON.stringify({ action: "verify", phone: lWaPhone, code: lWaOtp }),
   243→      });
   244→      const data = await res.json();
   245→      if (!res.ok) {
   246→        toast.error(data.error || tr("regOtpInvalid"));
   247→        return;
   248→      }
   249→      toast.success(tr("loginWaVerified"));
   250→      setLWaOtpVerified(true);
   251→    } catch {
   252→      toast.error(tr("errConnection"));
   253→    } finally {
   254→      setLWaOtpVerifying(false);
   255→    }
   256→  };
   257→
   258→  const doLogin = async (e: React.FormEvent) => {
   259→    e.preventDefault();
   260→
   261→    // ===== WhatsApp OTP login path =====
   262→    if (loginMethod === "wa") {
   263→      if (!lWaPhone.trim()) {
   264→        toast.error(tr("errPhoneRequired"));
   265→        return;
   266→      }
   267→      if (!lWaOtpVerified) {
   268→        toast.error(tr("loginWaOtpFirst"));
   269→        return;
   270→      }
   271→      setLoading(true);
   272→      try {
   273→        const res = await fetch("/api/auth/login", {
   274→          method: "POST",
   275→          headers: { "Content-Type": "application/json" },
   276→          body: JSON.stringify({ phone: lWaPhone }),
   277→        });
   278→        const data = await res.json();
   279→        if (!res.ok) {
   280→          const isNotFound = /tidak terdaftar|not registered|not found/i.test(data.error || "");
   281→          toast.error(isNotFound ? tr("loginWaNotFound") : (data.error || tr("errLogin")));
   282→          return;
   283→        }
   284→        const setUser = useStore.getState().setUser;
   285→        const goHome = useStore.getState().goHome;
   286→        const goToAdmin = useStore.getState().goToAdmin;
   287→        setUser(data.user);
   288→        if (data.sessionToken) setSessionToken(data.sessionToken);
   289→        setSuccess(true);
   290→        toast.success(formatT(tr("welcomeBack"), { name: data.user.name }));
   291→        const isAdmin = data.user.role === "admin" || data.user.role === "superadmin";
   292→        setTimeout(() => isAdmin ? goToAdmin() : goHome(), 900);
   293→      } catch {
   294→        toast.error(tr("errConnection"));
   295→      } finally {
   296→        setLoading(false);
   297→      }
   298→      return;
   299→    }
   300→
   301→    // ===== Email + password login path (default) =====
   302→    if (!lEmail.trim() || !lPass) {
   303→      toast.error(tr("errEmailPass"));
   304→      return;
   305→    }
   306→    setLoading(true);
   307→    try {
   308→      const res = await fetch("/api/auth/login", {
   309→        method: "POST",
   310→        headers: { "Content-Type": "application/json" },
   311→        body: JSON.stringify({ email: lEmail, password: lPass }),
   312→      });
   313→      const data = await res.json();
   314→      if (!res.ok) {
   315→        toast.error(data.error || tr("errLogin"));
   316→        return;
   317→      }
   318→      const setUser = useStore.getState().setUser;
   319→      const goHome = useStore.getState().goHome;
   320→      const goToAdmin = useStore.getState().goToAdmin;
   321→      setUser(data.user);
   322→      if (data.sessionToken) setSessionToken(data.sessionToken);
   323→      setSuccess(true);
   324→      toast.success(formatT(tr("welcomeBack"), { name: data.user.name }));
   325→      const isAdmin = data.user.role === "admin" || data.user.role === "superadmin";
   326→      setTimeout(() => isAdmin ? goToAdmin() : goHome(), 900);
   327→    } catch {
   328→      toast.error(tr("errConnection"));
   329→    } finally {
   330→      setLoading(false);
   331→    }
   332→  };
   333→
   334→  const sendRegOtp = async () => {
   335→    if (!rPhone.trim()) {
   336→      toast.error(tr("regOtpPhoneFirst"));
   337→      return;
   338→    }
   339→    if (rPhoneStatus === "taken") {
   340→      toast.error(tr("errPhoneTaken"));
   341→      return;
   342→    }
   343→    if (rPhoneStatus === "checking") {
   344→      toast.error(tr("phoneChecking"));
   345→      return;
   346→    }
   347→    setROtpSending(true);
   348→    setROtpDevCode(null);
   349→    try {
   350→      const res = await fetch("/api/auth/register-otp", {
   351→        method: "POST",
   352→        headers: { "Content-Type": "application/json" },
   353→        body: JSON.stringify({ action: "send", phone: rPhone }),
   354→      });
   355→      const data = await res.json();
   356→      if (!res.ok) {
   357→        toast.error(data.error || tr("errConnection"));
   358→        return;
   359→      }
   360→      toast.success(data.message || tr("regOtpSent"));
   361→      if (data._devCode) {
   362→        setROtpDevCode(data._devCode);
   363→      }
   364→      setROtpCooldown(60);
   365→    } catch {
   366→      toast.error(tr("errConnection"));
   367→    } finally {
   368→      setROtpSending(false);
   369→    }
   370→  };
   371→
   372→  const verifyRegOtp = async () => {
   373→    if (rOtp.length < 6) {
   374→      toast.error(tr("regOtpRequired"));
   375→      return;
   376→    }
   377→    setROtpVerifying(true);
   378→    try {
   379→      const res = await fetch("/api/auth/register-otp", {
   380→        method: "POST",
   381→        headers: { "Content-Type": "application/json" },
   382→        body: JSON.stringify({ action: "verify", phone: rPhone, code: rOtp }),
   383→      });
   384→      const data = await res.json();
   385→      if (!res.ok) {
   386→        toast.error(data.error || tr("regOtpInvalid"));
   387→        return;
   388→      }
   389→      toast.success(tr("otpVerified"));
   390→      setROtpVerified(true);
   391→    } catch {
   392→      toast.error(tr("errConnection"));
   393→    } finally {
   394→      setROtpVerifying(false);
   395→    }
   396→  };
   397→
   398→  // When phone changes after verification, reset verified state so user must re-verify.
   399→  const prevPhoneRef = useRef<string>("");
   400→  useEffect(() => {
   401→    if (prevPhoneRef.current !== rPhone) {
   402→      prevPhoneRef.current = rPhone;
   403→      if (rOtpVerified) setROtpVerified(false);
   404→    }
   405→  }, [rPhone, rOtpVerified]);
   406→
   407→  const doRegister = async (e: React.FormEvent) => {
   408→    e.preventDefault();
   409→    if (!rName.trim() || !rEmail.trim() || !rPass) {
   410→      toast.error(tr("errRequired"));
   411→      return;
   412→    }
   413→    if (rEmailStatus === "invalid") {
   414→      toast.error(tr("emailInvalid"));
   415→      return;
   416→    }
   417→    if (rEmailStatus === "taken") {
   418→      toast.error(tr("errEmailTaken"));
   419→      return;
   420→    }
   421→    // If still checking, wait for it to resolve before allowing registration
   422→    // so we never submit a duplicate even in a race.
   423→    if (rEmailStatus === "checking") {
   424→      toast.error(tr("emailChecking"));
   425→      return;
   426→    }
   427→    if (rNameStatus === "taken") {
   428→      toast.error(tr("errNameTaken"));
   429→      return;
   430→    }
   431→    if (rNameStatus === "checking") {
   432→      toast.error(tr("nameChecking"));
   433→      return;
   434→    }
   435→    if (rPhoneStatus === "taken") {
   436→      toast.error(tr("errPhoneTaken"));
   437→      return;
   438→    }
   439→    if (rPhoneStatus === "checking") {
   440→      toast.error(tr("phoneChecking"));
   441→      return;
   442→    }
   443→    if (!rOtpVerified) {
   444→      toast.error(tr("errPhoneNotVerified"));
   445→      return;
   446→    }
   447→    if (rPass.length < 6) {
   448→      toast.error(tr("errPassLength"));
   449→      return;
   450→    }
   451→    if (rPass !== rPass2) {
   452→      toast.error(tr("errPassMatch"));
   453→      return;
   454→    }
   455→    if (!agree) {
   456→      toast.error(tr("errAgree"));
   457→      return;
   458→    }
   459→    setLoading(true);
   460→    try {
   461→      const res = await fetch("/api/auth/register", {
   462→        method: "POST",
   463→        headers: { "Content-Type": "application/json" },
   464→        body: JSON.stringify({
   465→          name: rName,
   466→          email: rEmail,
   467→          password: rPass,
   468→          phone: rPhone,
   469→        }),
   470→      });
   471→      const data = await res.json();
   472→      if (!res.ok) {
   473→        toast.error(data.error || tr("errRegister"));
   474→        return;
   475→      }
   476→      const setUser = useStore.getState().setUser;
   477→      const goToPost = useStore.getState().goToPost;
   478→      const goToAdmin = useStore.getState().goToAdmin;
   479→      setUser(data.user);
   480→      if (data.sessionToken) setSessionToken(data.sessionToken);
   481→      setSuccess(true);
   482→      toast.success(tr("registerSuccess"));
   483→      const isAdmin = data.user.role === "admin" || data.user.role === "superadmin";
   484→      setTimeout(() => isAdmin ? goToAdmin() : goToPost(), 1100);
   485→    } catch {
   486→      toast.error(tr("errConnection"));
   487→    } finally {
   488→      setLoading(false);
   489→    }
   490→  };
   491→
   492→  if (success) {
   493→    return (
   494→      <div className="mx-auto flex max-w-md flex-col items-center px-4 py-20 text-center animate-fade-up">
   495→        <div className="grid size-20 place-items-center rounded-full bg-primary/10">
   496→          <CheckCircle2 className="size-12 text-primary" />
   497→        </div>
   498→        <h2 className="mt-4 text-2xl font-bold">{tr("loginSuccess")}</h2>
   499→        <p className="mt-2 text-sm text-muted-foreground">
   500→          {tr("loginRedirect")}
   501→        </p>
   502→        <Loader2 className="mt-4 size-5 animate-spin text-primary" />
   503→      </div>
   504→    );
   505→  }
   506→
   507→  return (
   508→    <div className="animate-fade-up">
   509→      {/* ===== MOBILE: compact single column ===== */}
   510→      <div className="flex min-h-[calc(100vh-4rem)] flex-col px-4 py-6 md:hidden">
   511→        <button
   512→          onClick={goBack}
   513→          className="mb-4 inline-flex w-fit items-center gap-1 text-sm text-muted-foreground hover:text-primary"
   514→        >
   515→          <ChevronLeft className="size-4" /> {tr("back")}
   516→        </button>
   517→        <div className="flex flex-col items-center text-center">
   518→          {/* eslint-disable-next-line @next/next/no-img-element */}
   519→          <img src="/logo.jpeg" alt="mesinKU" className="size-16 rounded-2xl shadow-lg object-cover" />
   520→          <h1 className="mt-3 text-2xl font-extrabold tracking-tight">
   521→            <span className="text-primary">mesin</span>KU
   522→          </h1>
   523→          <p className="mt-1 text-sm text-muted-foreground">{tr("loginTitle")}</p>
   524→        </div>
   525→        <div className="mt-6 w-full max-w-md mx-auto">
   526→          <FormSection
   527→            tab={tab} setTab={setTab}
   528→            showPass={showPass} setShowPass={setShowPass}
   529→            loading={loading}
   530→            lEmail={lEmail} setLEmail={setLEmail}
   531→            lPass={lPass} setLPass={setLPass}
   532→            loginMethod={loginMethod} setLoginMethod={setLoginMethod}
   533→            lWaPhone={lWaPhone} setLWaPhone={setLWaPhone}
   534→            lWaOtp={lWaOtp} setLWaOtp={setLWaOtp}
   535→            lWaOtpSending={lWaOtpSending}
   536→            lWaOtpVerifying={lWaOtpVerifying}
   537→            lWaOtpVerified={lWaOtpVerified}
   538→            lWaOtpDevCode={lWaOtpDevCode}
   539→            lWaCooldown={lWaCooldown}
   540→            sendLoginOtp={sendLoginOtp}
   541→            verifyLoginOtp={verifyLoginOtp}
   542→            rName={rName} setRName={setRName}
   543→            rNameStatus={rNameStatus}
   544→            rEmail={rEmail} setREmail={setREmail}
   545→            rEmailStatus={rEmailStatus}
   546→            rPhone={rPhone} setRPhone={setRPhone}
   547→            rPhoneStatus={rPhoneStatus}
   548→            rPass={rPass} setRPass={setRPass}
   549→            rPass2={rPass2} setRPass2={setRPass2}
   550→            agree={agree} setAgree={setAgree}
   551→            rOtp={rOtp} setROtp={setROtp}
   552→            rOtpSending={rOtpSending}
