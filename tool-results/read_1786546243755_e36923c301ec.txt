    35→// Chat image with error fallback — handles expired tmpfiles.org URLs.
    36→function ChatBubbleImage({ src, onLightbox }: { src: string; onLightbox?: (url: string) => void }) {
    37→  const [err, setErr] = useState(false);
    38→  const url = normalizeImageUrl(src);
    39→  if (err) {
    40→    return (
    41→      <a
    42→        href={src}
    43→        target="_blank"
    44→        rel="noopener noreferrer"
    45→        onClick={(e) => { e.preventDefault(); onLightbox?.(src); }}
    46→        className="flex min-w-[180px] items-center gap-2 bg-black/5 px-3 py-4 text-xs underline opacity-80"
    47→      >
    48→        <FileImage className="size-4 shrink-0" />
    49→        Gambar tidak tersedia — klik untuk membuka
    50→      </a>
    51→    );
    52→  }
    53→  return (
    54→    <img
    55→      src={url}
    56→      alt="Gambar"
    57→      referrerPolicy="no-referrer"
    58→      onClick={() => onLightbox?.(src)}
    59→      onError={() => setErr(true)}
    60→      className="block w-full min-w-[180px] max-w-full cursor-pointer object-cover transition hover:opacity-90"
    61→      style={{ maxHeight: 300 }}
    62→    />
    63→  );
    64→}

    74→/**
    75→ * ChatInner — the reusable chat UI (header, listing card, messages, input,
    76→ * context menu, lightbox). Used by BOTH:
    77→ *   - ChatWidget: wraps ChatInner in a Dialog (modal)
    78→ *   - ChatView:   wraps ChatInner in a full-page container
    79→ *
    80→ * Because Radix Dialog only mounts DialogContent children when open, and
    81→ * ChatView only renders when view === "chat", ChatInner is ALWAYS "active"
    82→ * while mounted — no `open` prop needed.
    83→ */
    84→export function ChatInner({
    85→  listing,
    86→  onBack,
    87→  variant = "modal",
    88→}: {
    89→  listing: Listing;
    90→  onBack: () => void;
    91→  variant?: "modal" | "page";
    92→}) {
    93→  const [messages, setMessages] = useState<Msg[]>([]);
    94→  const [input, setInput] = useState("");
    95→  const [sending, setSending] = useState(false);
    96→  const [loadedHistory, setLoadedHistory] = useState(false);
    97→  const [menu, setMenu] = useState<MenuState>({ visible: false, x: 0, y: 0, msgIndex: -1 });
    98→  const [selectedMsg, setSelectedMsg] = useState<number | null>(null);
    99→  const [blocked, setBlocked] = useState(false);
   100→  const [imgSending, setImgSending] = useState(false);
   101→  const [bgOpen, setBgOpen] = useState(false);
   102→  const [lightbox, setLightbox] = useState<string | null>(null);
   103→  const scrollRef = useRef<HTMLDivElement>(null);
   104→  const longPressTimer = useRef<NodeJS.Timeout | null>(null);
   105→  const imgFileRef = useRef<HTMLInputElement>(null);
   106→  const imgCamRef = useRef<HTMLInputElement>(null);
   107→  const { t } = useLang();
   108→  const mounted = useMounted();
   109→  const tr = mounted ? t : (key: any) => (i18nTranslations.id as any)[key] ?? key;
   110→  const quick = [tr("quick1"), tr("quick2"), tr("quick3"), tr("quick4")];
   111→
   112→  const seller: Seller = listing.seller;
   113→  const ownerId = (listing as any).user?.id || null;
   114→  const ownerName = (listing as any).user?.name || seller.name;
   115→  const currentUser = useStore((s) => s.user);
   116→  const goToLogin = useStore((s) => s.goToLogin);
   117→  const { bgKey, setBg, bgStyle, isDark: bgIsDark, presets } = useChatBg();
   118→
   119→  const queryClient = useQueryClient();
   120→  const { sendMessage, markRead, subscribe } = useChatSocket();
   121→
   122→  // Fetch conversation history. Poll every 5s so messages sent via HTTP
   123→  // fallback (e.g. on Vercel where socket.io isn't running) still appear.
   124→  const { data: historyData } = useQuery({
   125→    queryKey: ["chat-history", currentUser?.id, ownerId, listing.id],
   126→    queryFn: async () => {
   127→      if (!currentUser || !ownerId) return { conversations: [] };
   128→      const res = await fetch(`/api/messages?userId=${currentUser.id}`);
   129→      if (!res.ok) throw new Error("fail");
   130→      return res.json();
   131→    },
   132→    enabled: !!currentUser && !!ownerId,
   133→    refetchInterval: 5000,
   134→    refetchIntervalInBackground: false,
   135→  });
   136→
   137→  // Merge history data into local messages state. On the FIRST load we
   138→  // replace the entire list; on subsequent polls we MERGE by id so that we
   139→  // don't lose optimistic local messages and don't cause duplicate renders.
   140→  useEffect(() => {
   141→    if (!historyData?.conversations) return;
   142→    const conv = historyData.conversations.find(
   143→      (c: any) => c.partnerId === ownerId && c.listingTitle === listing.title
   144→    );
   145→    const dbMsgs: Msg[] = conv?.messages?.length
   146→      ? [...conv.messages].reverse().map((m: any) => ({
   147→          id: m.id,
   148→          role: m.sent ? "user" : "assistant",
   149→          content: m.content,
   150→          image: m.image || null,
   151→          time: new Date(m.createdAt).toLocaleTimeString("id-ID", { hour: "2-digit", minute: "2-digit" }),
   152→        }))
   153→      : [];
   154→
   155→    if (!loadedHistory) {
   156→      setMessages(dbMsgs);
   157→      setLoadedHistory(true);
   158→      if (currentUser && ownerId && conv?.messages?.length) {
   159→        markRead(currentUser.id, ownerId);
   160→      }
   161→    } else {
   162→      setMessages((prev) => {
   163→        const existingIds = new Set(prev.filter((m) => m.id).map((m) => m.id));
   164→        const newOnes = dbMsgs.filter((m) => m.id && !existingIds.has(m.id));
   165→        if (newOnes.length === 0) return prev;
   166→        const hasIncoming = newOnes.some((m) => m.role === "assistant");
   167→        if (hasIncoming && currentUser && ownerId) {
   168→          markRead(currentUser.id, ownerId);
   169→        }
   170→        return [...prev, ...newOnes];
   171→      });
   172→    }
   173→  }, [historyData, ownerId, listing.title, loadedHistory, currentUser, markRead]);
   174→
   175→  useEffect(() => {
   176→    if (!currentUser || !ownerId) return;
   177→    const off = subscribe<ChatMessage>("message:new", (msg) => {
   178→      const isRelevant =
   179→        (msg.sent === false && msg.senderId === ownerId && msg.receiverId === currentUser.id) ||
   180→        (msg.sent === true && msg.senderId === currentUser.id && msg.receiverId === ownerId);
   181→      if (!isRelevant) return;
   182→      const time = new Date(msg.createdAt).toLocaleTimeString("id-ID", { hour: "2-digit", minute: "2-digit" });
   183→      setMessages((prev) => {
   184→        const last = prev[prev.length - 1];
   185→        if (last && last.role === (msg.sent ? "user" : "assistant") && last.content === msg.content && last.time === time) {
   186→          return prev;
   187→        }
   188→        return [...prev, { id: msg.id, role: msg.sent ? "user" : "assistant", content: msg.content, image: (msg as any).image || null, time }];
   189→      });
   190→      if (!msg.sent) {
   191→        markRead(currentUser.id, ownerId);
   192→        queryClient.invalidateQueries({ queryKey: ["messages"] });
   193→      }
   194→    });
   195→    return off;
   196→  }, [currentUser, ownerId, subscribe, markRead, queryClient]);
   197→
   198→  // Track chat-open state so the global Header knows to play a soft "ding"
   199→  // (instead of the full ringtone) when a new message arrives while the user
   200→  // is already viewing this chat.
   201→  useEffect(() => {
   202→    setChatOpen(true);
   203→    return () => {
   204→      setChatOpen(false);
   205→    };
   206→  }, []);
   207→
   208→  useEffect(() => {
   209→    scrollRef.current?.scrollTo({ top: scrollRef.current.scrollHeight, behavior: "smooth" });
   210→  }, [messages, sending]);
   211→
   212→  // Close menu on click outside.
   213→  useEffect(() => {
   214→    if (!menu.visible) return;
   215→    const close = () => setMenu({ visible: false, x: 0, y: 0, msgIndex: -1 });
   216→    window.addEventListener("click", close);
   217→    window.addEventListener("contextmenu", close);
   218→    return () => {
   219→      window.removeEventListener("click", close);
   220→      window.removeEventListener("contextmenu", close);
   221→    };
   222→  }, [menu.visible]);
   223→
   224→  const send = async (text: string) => {
   225→    const content = text.trim();
   226→    if (!content || sending || blocked) return;
   227→    if (!currentUser) {
   228→      toast.info(tr("chatLoginRequired"), { action: { label: tr("chatLoginAction"), onClick: goToLogin } });
   229→      return;
   230→    }
   231→    if (!ownerId) { toast.info(tr("chatSellerNotRegistered")); return; }
   232→    if (currentUser.id === ownerId) { toast.info(tr("chatOwnListing")); return; }
   233→
   234→    setInput("");
   235→    const now = new Date().toLocaleTimeString("id-ID", { hour: "2-digit", minute: "2-digit" });
   236→    setMessages(prev => [...prev, { role: "user", content, time: now }]);
   237→    setSending(true);
   238→
   239→    try {
   240→      const ack = await sendMessage({
   241→        senderId: currentUser.id, receiverId: ownerId, content,
   242→        listingId: listing.id, listingTitle: listing.title,
   243→      });
   244→      if (!ack?.ok) {
   245→        await fetch("/api/messages", {
   246→          method: "POST", headers: { "Content-Type": "application/json" },
   247→          body: JSON.stringify({ senderId: currentUser.id, receiverId: ownerId, content, listingId: listing.id, listingTitle: listing.title }),
   248→        });
   249→      }
   250→      queryClient.invalidateQueries({ queryKey: ["messages"] });
   251→      queryClient.invalidateQueries({ queryKey: ["chat-history"] });
   252→    } catch { toast.error(tr("chatSendFailed")); }
   253→    finally { setSending(false); }
   254→  };
   255→
   256→  // ===== Context menu actions =====
   257→  const handleDeleteMessage = useCallback(async (index: number) => {
   258→    const msg = messages[index];
   259→    if (!msg) return;
   260→    setMessages(prev => prev.filter((_, i) => i !== index));
   261→    setMenu({ visible: false, x: 0, y: 0, msgIndex: -1 });
   262→    setSelectedMsg(null);
   263→    if (msg.id) {
   264→      try {
   265→        await fetch("/api/messages", { method: "DELETE", headers: { "Content-Type": "application/json" }, body: JSON.stringify({ messageId: msg.id }) });
   266→        toast.success("Pesan dihapus");
   267→      } catch { toast.error("Gagal menghapus pesan"); }
   268→    }
   269→  }, [messages]);
   270→
   271→  const handleClearChat = useCallback(async () => {
   272→    if (!currentUser || !ownerId) return;
   273→    setMessages([]);
   274→    setMenu({ visible: false, x: 0, y: 0, msgIndex: -1 });
   275→    setSelectedMsg(null);
   276→    try {
   277→      await fetch("/api/messages", { method: "DELETE", headers: { "Content-Type": "application/json" }, body: JSON.stringify({ userId: currentUser.id, partnerId: ownerId, listingTitle: listing.title }) });
   278→      toast.success("Chat dibersihkan");
   279→    } catch { toast.error("Gagal membersihkan chat"); }
   280→  }, [currentUser, ownerId, listing.title]);
   281→
   282→  const handleBlock = useCallback(() => {
   283→    setBlocked(true);
   284→    setMenu({ visible: false, x: 0, y: 0, msgIndex: -1 });
   285→    setSelectedMsg(null);
   286→    toast.success(`${ownerName} diblokir`, { description: "Anda tidak akan menerima pesan dari pengguna ini." });
   287→  }, [ownerName]);
   288→
   289→  // ===== Desktop: right-click context menu =====
   290→  const handleContextMenu = (e: React.MouseEvent, index: number) => {
   291→    e.preventDefault();
   292→    e.stopPropagation();
   293→    const rect = e.currentTarget.getBoundingClientRect();
   294→    setMenu({ visible: true, x: e.clientX, y: e.clientY, msgIndex: index });
   295→    setSelectedMsg(index);
   296→  };
   297→
   298→  // ===== Mobile: long-press =====
   299→  const handleTouchStart = (index: number) => {
   300→    longPressTimer.current = setTimeout(() => {
   301→      setSelectedMsg(index);
   302→      const el = document.querySelector(`[data-msg-index="${index}"]`) as HTMLElement;
   303→      if (el) {
   304→        const rect = el.getBoundingClientRect();
   305→        setMenu({ visible: true, x: rect.left + rect.width / 2, y: rect.top + rect.height / 2, msgIndex: index });
   306→      }
   307→    }, 500);
   308→  };
   309→
   310→  const handleTouchEnd = () => {
   311→    if (longPressTimer.current) {
   312→      clearTimeout(longPressTimer.current);
   313→      longPressTimer.current = null;
   314→    }
   315→  };
   316→
   317→  const handleChatImage = async (e: React.ChangeEvent<HTMLInputElement>) => {
   318→    const files = e.target.files;
   319→    if (!files || files.length === 0) return;
   320→    if (!currentUser) {
   321→      toast.info(tr("chatLoginRequired"), { action: { label: tr("chatLoginAction"), onClick: goToLogin } });
   322→      e.target.value = "";
   323→      return;
   324→    }
   325→    if (!ownerId) { toast.info(tr("chatSellerNotRegistered")); e.target.value = ""; return; }
   326→    if (currentUser.id === ownerId) { toast.info(tr("chatOwnListing")); e.target.value = ""; return; }
   327→    setImgSending(true);
   328→    try {
   329→      for (const file of Array.from(files)) {
   330→        const compressed = await compressImage(file);
   331→        const now = new Date().toLocaleTimeString("id-ID", { hour: "2-digit", minute: "2-digit" });
   332→        const localMsg: Msg = { role: "user", content: "📷 Foto", image: compressed, time: now };
   333→        setMessages((prev) => [...prev, localMsg]);
   334→        try {
   335→          const ack = await sendMessage({
   336→            senderId: currentUser.id, receiverId: ownerId, content: "📷 Foto",
   337→            image: compressed, listingId: listing.id, listingTitle: listing.title,
   338→          });
   339→          if (!ack?.ok) {
   340→            await fetch("/api/messages", {
   341→              method: "POST", headers: { "Content-Type": "application/json" },
   342→              body: JSON.stringify({ senderId: currentUser.id, receiverId: ownerId, content: "📷 Foto", image: compressed, listingId: listing.id, listingTitle: listing.title }),
   343→            });
   344→          }
   345→          queryClient.invalidateQueries({ queryKey: ["messages"] });
   346→          queryClient.invalidateQueries({ queryKey: ["chat-history"] });
   347→        } catch { toast.error("Gagal mengirim foto"); }
   348→      }
   349→    } catch (err: any) {
   350→      toast.error(err?.message || "Gagal memproses gambar");
   351→    } finally {
   352→      setImgSending(false);
   353→      e.target.value = "";
   354→    }
   355→  };
   356→
   357→  const initials = ownerName.split(" ").slice(0, 2).map((w) => w[0]).join("").toUpperCase();
   358→  const img = listing.images?.[0];
   359→
   360→  const menuItems = menu.msgIndex >= 0 ? [
   361→    { label: "Hapus Pesan", icon: Trash2, action: () => handleDeleteMessage(menu.msgIndex), color: "text-red-600" },
   362→    { label: "Blokir Pengguna", icon: Ban, action: handleBlock, color: "text-red-600" },
   363→    { label: "Bersihkan Chat", icon: Eraser, action: handleClearChat, color: "text-foreground" },
   364→  ] : [];
   365→
   366→  // Page variant uses a slimmer header (sticky bar) so the full-page chat
   367→  // looks like a native messaging app rather than a modal.
   368→  const isPage = variant === "page";
   369→
   370→  return (
   371→    <>
   372→      {/* ===== HEADER ===== */}
   373→      <div className={cn(
   374→        "flex items-center gap-2 border-b border-border bg-primary p-3 text-primary-foreground",
   375→        isPage && "sticky top-0 z-10"
   376→      )}>
   377→        <button
   378→          onClick={onBack}
   379→          aria-label="Kembali"
   380→          className="grid size-8 shrink-0 place-items-center rounded-full text-primary-foreground transition hover:bg-white/15"
   381→        >
   382→          <ArrowLeft className="size-5" />
   383→        </button>
   384→        <Avatar className="size-10 border-2 border-white/30">
   385→          <AvatarFallback className="bg-white/20 text-sm font-bold text-primary-foreground">{initials}</AvatarFallback>
   386→        </Avatar>
   387→        <div className="min-w-0 flex-1">
   388→          <div className="flex items-center gap-1.5 text-sm text-primary-foreground">
   389→            <span className="truncate font-semibold">{ownerName}</span>
   390→            {seller.verified && <BadgeCheck className="size-3.5 shrink-0" />}
   391→            {blocked && <span className="rounded bg-red-500 px-1 text-[9px] font-bold">DIBLOKIR</span>}
   392→          </div>
   393→          <div className="flex items-center gap-1 text-[11px] text-primary-foreground/80">
   394→            <span className={cn("size-1.5 rounded-full", blocked ? "bg-red-400" : "bg-orange-400")} /> {blocked ? "Diblokir" : tr("chatOnline")}
   395→          </div>
   396→        </div>
   397→        <Popover open={bgOpen} onOpenChange={setBgOpen}>
   398→          <PopoverTrigger asChild>
   399→            <button
   400→              className="grid size-8 shrink-0 place-items-center rounded-full text-primary-foreground transition hover:bg-white/15"
   401→              aria-label="Pengaturan chat"
   402→            >
   403→              <Settings className="size-5" />
   404→            </button>
   405→          </PopoverTrigger>
   406→          <PopoverContent className="w-64 p-3" align="end">
   407→            <p className="mb-2 text-xs font-bold text-foreground">Warna Background Chat</p>
   408→            <div className="grid grid-cols-5 gap-2">
   409→              {presets.map((p) => (
   410→                <button
   411→                  key={p.key}
   412→                  onClick={() => { setBg(p.key); }}
   413→                  className={cn(
   414→                    "size-9 rounded-full border-2 transition",
   415→                    bgKey === p.key ? "border-primary ring-2 ring-primary/30" : "border-border hover:scale-110"
   416→                  )}
   417→                  style={{ backgroundColor: p.color }}
   418→                  title={p.label}
   419→                  aria-label={p.label}
   420→                />
   421→              ))}
   422→            </div>
   423→          </PopoverContent>
   424→        </Popover>
   425→      </div>
   426→
   427→      {/* ===== LISTING CARD ===== */}
   428→      <div className="shrink-0 border-b border-border bg-card p-2.5">
   429→        <div className="flex items-center gap-2.5">
   430→          <div className="relative size-[60px] shrink-0 overflow-hidden rounded-lg bg-muted sm:size-[80px]">
   431→            {img ? <img src={img} alt="" className="size-full object-cover" /> : <div className="flex h-full items-center justify-center text-muted-foreground"><Tag className="size-6" /></div>}
   432→          </div>
   433→          <div className="min-w-0 flex-1">
   434→            <p className="line-clamp-2 text-sm font-semibold text-foreground">{listing.title}</p>
   435→            <p className="mt-0.5 text-sm font-bold text-primary">{formatRupiahFull(listing.price)}</p>
   436→            <p className="flex items-center gap-0.5 text-[10px] text-muted-foreground"><MapPin className="size-2.5" /> {listing.city}</p>
   437→          </div>
   438→        </div>
   439→      </div>
   440→
   441→      {/* ===== MESSAGES ===== */}
   442→      <div
   443→        ref={scrollRef}
   444→        className={cn(
   445→          "gomesin-scroll min-h-0 flex-1 space-y-1 overflow-y-auto p-3",
   446→          bgIsDark && "text-white",
   447→          isPage ? "sm:max-h-none" : "max-h-[40vh] min-h-[180px]"
   448→        )}
   449→        style={bgStyle}
   450→      >
   451→        {messages.length === 0 && loadedHistory && (
   452→          <div className="flex flex-col items-center justify-center py-8 text-center">
   453→            <MessageCircle className="size-8 text-muted-foreground/40" />
   454→            <p className="mt-2 text-xs text-muted-foreground">Belum ada pesan. Mulai chat dengan penjual.</p>
   455→          </div>
   456→        )}
   457→        {messages.length > 0 && (
   458→          <div className="flex justify-center py-1">
   459→            <span className="rounded-full bg-white/80 px-3 py-0.5 text-[10px] font-medium text-black/60 shadow-sm">Hari ini</span>
   460→          </div>
   461→        )}
   462→        {messages.map((m, i) => (
   463→          <div key={i} className={m.role === "user" ? "flex justify-end" : "flex justify-start"}>
   464→            <div className={cn("flex flex-col gap-0.5", m.role === "user" ? "items-end" : "items-start")}>
   465→              <div
   466→                data-msg-index={i}
   467→                onContextMenu={(e) => handleContextMenu(e, i)}
   468→                onTouchStart={() => handleTouchStart(i)}
   469→                onTouchEnd={handleTouchEnd}
   470→                onTouchMove={handleTouchEnd}
   471→                className={cn(
   472→                  "max-w-[85%] cursor-pointer rounded-2xl text-sm shadow-sm transition select-none",
   473→                  m.role === "user"
   474→                    ? "rounded-br-sm bg-primary text-primary-foreground"
   475→                    : cn("rounded-bl-sm bg-white text-black font-medium"),
   476→                  selectedMsg === i && "ring-2 ring-blue-400 ring-offset-1",
   477→                  m.image ? "overflow-hidden p-0" : "px-3 py-2"
   478→                )}
   479→              >
   480→                {m.image && (
   481→                  <ChatBubbleImage src={m.image} onLightbox={(url) => setLightbox(url)} />
   482→                )}
   483→                {m.content && m.image && (
   484→                  <p className="whitespace-pre-wrap break-words px-2.5 pt-1.5">{m.content}</p>
   485→                )}
   486→                {m.content && !m.image && m.content}
   487→                {!m.content && !m.image && ""}
   488→                {m.time && (
   489→                  <span className={cn("inline-block text-[9px] opacity-60", m.image ? "block px-2.5 pb-1 pt-0.5 text-right" : "ml-2")}>{m.time}</span>
   490→                )}
   491→              </div>
   492→            </div>
   493→          </div>
   494→        ))}
   495→        {sending && (
   496→          <div className="flex justify-end">
   497→            <div className="rounded-2xl rounded-br-sm bg-primary px-3 py-2 text-sm text-primary-foreground shadow-sm">
   498→              <span className="size-2 animate-bounce rounded-full bg-white/70 [animation-delay:-0.3s] inline-block" />
   499→              <span className="size-2 animate-bounce rounded-full bg-white/70 [animation-delay:-0.15s] inline-block ml-0.5" />
   500→              <span className="size-2 animate-bounce rounded-full bg-white/70 inline-block ml-0.5" />
   501→            </div>
   502→          </div>
   503→        )}
   504→      </div>
   505→
   506→      {/* ===== QUICK REPLIES ===== */}
   507→      {messages.length === 0 && loadedHistory && !blocked && (
   508→        <div className="flex flex-wrap gap-1.5 border-t border-border px-3 py-2">
   509→          {quick.map((q) => (
   510→            <button key={q} onClick={() => send(q)} className="rounded-full border border-primary/30 bg-primary/5 px-3 py-1 text-xs text-primary hover:bg-primary/10">{q}</button>
   511→          ))}
   512→        </div>
   513→      )}
   514→
   515→      {/* ===== INPUT ===== */}
   516→      <form
   517→        onSubmit={(e) => { e.preventDefault(); send(input); }}
   518→        className="flex shrink-0 items-center gap-1.5 border-t border-border bg-card p-2.5"
   519→      >
   520→        {/* Hidden image inputs */}
   521→        <input ref={imgFileRef} type="file" accept="image/*" className="hidden" onChange={handleChatImage} />
   522→        <input ref={imgCamRef} type="file" accept="image/*" capture="environment" className="hidden" onChange={handleChatImage} />
   523→        <Button
   524→          type="button"
   525→          size="icon"
   526→          variant="ghost"
   527→          className="size-9 shrink-0 text-muted-foreground hover:text-primary"
   528→          disabled={imgSending || blocked}
   529→          onClick={() => imgFileRef.current?.click()}
   530→        >
   531→          {imgSending ? <Loader2 className="size-4 animate-spin" /> : <ImagePlus className="size-4" />}
   532→        </Button>
   533→        <Input
   534→          value={input}
   535→          onChange={(e) => setInput(e.target.value)}
   536→          placeholder={blocked ? "Pengguna diblokir" : tr("chatPlaceholder")}
   537→          className="h-9 flex-1 rounded-full"
   538→          disabled={sending || blocked}
   539→        />
   540→        <Button type="submit" size="icon" className="size-9 shrink-0 rounded-full bg-primary" disabled={sending || !input.trim() || blocked}>
   541→          {sending ? <Loader2 className="size-4 animate-spin" /> : <Send className="size-4" />}
   542→        </Button>
   543→      </form>
   544→
   545→      {/* ===== CONTEXT MENU (desktop right-click + mobile long-press) ===== */}
   546→      {menu.visible && (
   547→        <div
   548→          className="fixed z-[100] min-w-[160px] overflow-hidden rounded-lg border border-border bg-card py-1 shadow-xl animate-fade-up"
   549→          style={{ left: Math.min(menu.x, window.innerWidth - 180), top: Math.min(menu.y, window.innerHeight - 200) }}
   550→          onClick={(e) => e.stopPropagation()}
   551→        >
   552→          {menuItems.map((item, idx) => (
   553→            <button
   554→              key={idx}
   555→              onClick={() => { item.action(); }}
   556→              className={cn("flex w-full items-center gap-2 px-3 py-2 text-sm hover:bg-accent transition", item.color)}
   557→            >
   558→              <item.icon className="size-4" />
   559→              {item.label}
   560→            </button>
   561→          ))}
   562→        </div>
   563→      )}
   564→
   565→      {/* ===== IMAGE LIGHTBOX — click image to view full size ===== */}
   566→      {lightbox && (
   567→        <div
   568→          className="fixed inset-0 z-[200] flex items-center justify-center bg-black/90 p-4"
   569→          onClick={() => setLightbox(null)}
   570→        >
   571→          <button
   572→            aria-label="Tutup"
   573→            className="absolute right-4 top-4 grid size-10 place-items-center rounded-full bg-white/10 text-white hover:bg-white/20"
   574→          >
   575→            <X className="size-6" />
   576→          </button>
   577→          <img
   578→            src={normalizeImageUrl(lightbox)}
   579→            alt="Gambar besar"
   580→            referrerPolicy="no-referrer"
   581→            className="max-h-[90vh] max-w-full rounded-lg object-contain"
   582→            onClick={(e) => e.stopPropagation()}
   583→          />
   584→        </div>
   585→      )}
   586→    </>
   587→  );
   588→}
