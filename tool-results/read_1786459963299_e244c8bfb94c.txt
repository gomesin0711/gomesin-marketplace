   176→  // Current package label for listing summary

   245→          package: selectedPackage,
   246→          paymentMethod: needsPayment ? paymentMethod : undefined,

   259→      } else {
   260→        onOpenChange(false);
   261→      }
   262→    } catch (e: any) {
   263→      toast.error(e.message || "Gagal mengaktifkan paket");

   265→      setSubmitting(false);
   266→    }

   282→              type="button"
   283→              onClick={() => onOpenChange(false)}
   284→              className="grid size-10 shrink-0 place-items-center rounded-full border border-border bg-card hover:bg-accent"

   286→            >
   287→              <X className="size-5" />
   288→            </button>
   289→          </div>
   290→
   291→          {/* Content 2 kolom (desktop) / 1 kolom (mobile) */}
   292→          <div className="grid flex-1 gap-6 md:grid-cols-2 md:overflow-hidden">
   293→
   294→            {/* LEFT — gambar iklan + detail */}
   295→            <div className="space-y-3 md:overflow-y-auto md:pr-2">
   296→              <div className="overflow-hidden rounded-xl border border-border bg-card">
   297→                {/* Gambar utama iklan (besar) */}
   298→                <div className="relative aspect-video w-full overflow-hidden bg-muted">
   299→                  {listing.images?.[0] ? (
   300→                    <Image
   301→                      src={listing.images[0]}
   302→                      alt={listingTitle(listing, mounted ? lang : "id")}
   303→                      fill
   304→                      className="object-cover"
   305→                      unoptimized
   306→                      sizes="(max-width: 768px) 100vw, 50vw"
   307→                    />
   308→                  ) : (
   309→                    <div className="flex h-full items-center justify-center text-muted-foreground">
   310→                      <ImageIcon className="size-12" />
   311→                    </div>
   312→                  )}
   313→                </div>
   314→                {/* Detail iklan */}
   315→                <div className="space-y-2 p-4">
   316→                  <p className="line-clamp-2 text-base font-semibold text-foreground">

   325→                  <div className="flex flex-wrap items-center gap-1.5">
   326→                    <span
   327→                      className={cn(
   328→                        "inline-flex items-center gap-1 rounded px-2 py-0.5 text-[10px] font-bold uppercase text-white",
   329→                        statusInfo.color
   330→                      )}
   331→                    >
   332→                      <StatusIcon className="size-3" />
   333→                      {statusInfo.text}
   334→                    </span>
   335→                    <span className="inline-flex items-center rounded bg-secondary px-2 py-0.5 text-[10px] font-medium text-muted-foreground">
   336→                      Paket saat ini: {currentPkgLabel}
   337→                    </span>
   338→                  </div>
   339→                  {/* Info tambahan */}
   340→                  <div className="grid grid-cols-2 gap-x-3 gap-y-1 pt-2 text-xs">

  1062→ * Removes special chars that break URLs/filenames.
  1063→ */

  1065→  return name
  1066→    .trim()
  1067→    .replace(/\s+/g, "_")           // spaces → underscore
  1068→    .replace(/[@/\\?%*:|"<>]/g, "_") // special chars → underscore
  1069→    .replace(/_+/g, "_")              // collapse multiple underscores
  1070→    .substring(0, 50);               // max 50 chars
  1071→}
