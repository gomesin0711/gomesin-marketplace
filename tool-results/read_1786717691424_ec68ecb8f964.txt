    84→    queryKey: ["messages", user?.id],

    87→      if (!res.ok) throw new Error("fail");

   101→  // The socket "message:new" handler (below) only fires when socket.io is
   102→  // connected. In production (Vercel, no socket server) OR when the socket
   103→  // is temporarily down, messages arrive only via the 3s polling above.
   104→  // This effect watches the polled messagesData and plays the notification

   106→  // the previous fetch. This guarantees the ringtone plays in ALL
   107→  // environments.
   108→  const seenMsgIdsRef = useRef<Set<string>>(new Set());

   111→  const initialLoadDoneRef = useRef(false);
   112→  useEffect(() => {
   113→    if (!user || !messagesData?.conversations) return;
   114→    const convs = messagesData.conversations as any[];
   115→    // Collect ALL incoming (received) message ids currently visible.
   116→    const currentIncomingIds: string[] = [];
   117→    for (const c of convs) {
   118→      for (const m of c.messages || []) {
   119→        if (m.sent === false) currentIncomingIds.push(m.id);
   120→      }
   121→    }
   122→    if (!initialLoadDoneRef.current) {
   123→      // First load: seed the seen set with all existing incoming ids so we
   124→      // don't play sound for messages that were already there.
   125→      currentIncomingIds.forEach((id) => seenMsgIdsRef.current.add(id));
   126→      initialLoadDoneRef.current = true;
   127→      return;
   128→    }
   129→    // Subsequent loads: find incoming ids that are NOT in the seen set.
   130→    const newIncoming = currentIncomingIds.filter((id) => !seenMsgIdsRef.current.has(id));
   131→    if (newIncoming.length === 0) return;
   132→    // Add them to the seen set.
   133→    newIncoming.forEach((id) => seenMsgIdsRef.current.add(id));

   135→    // chat, play a soft ding; otherwise play the full "mesinKU!" ringtone.

   137→      playDingSound();
   138→    } else {
   139→      playNotificationSound();
   140→    }

   151→      // ringtone so the user is alerted.
   152→      if (msg && msg.senderId !== user.id) {
   153→        // Mark this message id as seen immediately so the polling-based sound
   154→        // effect (above) doesn't double-play the same message after the
   155→        // refetch lands.
   156→        if (msg.id) seenMsgIdsRef.current.add(msg.id);
   157→        if (isChatOpen()) {
   158→          playDingSound();
   159→        } else {
   160→          playNotificationSound();
   161→        }
   162→      }
   163→    });
   164→    const offRead = subscribe("message:read-update", () => {
   165→      queryClient.invalidateQueries({ queryKey: ["messages"] });
   166→    });

   168→      offNew();
   169→      offRead();
   170→    };
   171→  }, [user, subscribe, queryClient]);

   380→        <div className="flex h-14 items-center gap-1 overflow-hidden px-3">
   381→          <Logo />
   382→          <div className="ml-auto flex items-center gap-0.5">
   383→            <Popover open={langOpenMobile} onOpenChange={setLangOpenMobile}>
   384→              <PopoverTrigger asChild>

   386→                  className="flex items-center gap-0.5 rounded-lg px-1.5 py-1.5 text-xs font-bold text-foreground hover:bg-accent"
   387→                  aria-label="Bahasa"
   388→                  suppressHydrationWarning

   392→              </PopoverTrigger>
   393→              <PopoverContent className="w-36 p-1" align="end">
   394→                <button
