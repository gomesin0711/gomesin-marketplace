    72→async function postListing(payload: any) {
    73→  const res = await fetch("/api/listings", {
    74→    method: "POST",
    75→    headers: { "Content-Type": "application/json" },
    76→    body: JSON.stringify(payload),
    77→    // NOTE: do NOT use keepalive:true here. The Fetch API spec limits
    78→    // keepalive request bodies to 64KB. Listing payloads contain base64-
    79→    // compressed images (~200KB each), so the payload easily exceeds 64KB.
    80→    // With keepalive:true, the browser throws "Failed to fetch" before the
    81→    // request even reaches the server.
    82→    //
    83→    // keepalive is no longer needed because doSubmit() is called LAST in the
    84→    // "Kirim & Pasang Iklan" flow (after upload-proof, chat, and WhatsApp).
    85→    // mutation.onSuccess (which navigates to "Iklan Saya") only fires AFTER
    86→    // the POST response returns, so the page does not navigate away mid-fetch.
    87→  });
    88→  const data = await res.json();
    89→  if (!res.ok) throw new Error(data.error || tr("postFailed"));
    90→  return data.listing;
    91→}
